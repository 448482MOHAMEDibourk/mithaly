> (Awareness Context) يتضمن:

> 3. موافقة السياسات (Policy Approval)"

## التطبيق البرمجي:

```python
# src/mithaly/core/execution/safe_executor.py

class SafeExecutor:
    def process(self, payload: Dict[str, Any]) -> Dict[str, Any]:
        """Execute plan with mandatory awareness check."""
        
        # ═══ CONSTITUTIONAL ENFORCEMENT ═══
        # المادة الخامسة: حتمية الإدراك
        if 'probe_results' not in payload:
            raise ConstitutionalViolation(
                article="المادة الخامسة",
                violation="إصدار أمر تنفيذي بدون تقرير وعي",
                remedy="يجب استدعاء probe() قبل execution"
            )
        
        probes = payload['probe_results']
        
        # فحص المخاطر من الطبقات السابقة
        risk_level = self._assess_risk(probes)
        
        if risk_level == 'CRITICAL':
            # تفعيل الانعطافية
            return {
                'status': 'INFLECTION_TRIGGERED',
                'reason': 'مخاطر حرجة مكتشفة من layer probes',
                'probes': probes
            }
        
        # التنفيذ الآمن (مع الوعي)
        return self._execute_aware(payload['plan'], probes)
    
    def _assess_risk(self, probes: Dict) -> str:
        """تقييم المخاطر من نتائج الـ probes."""
        # دمج نتائج الطبقات
        gaps_probe = probes.get('gaps', {})
        planning_probe = probes.get('planning', {})
        
        # منطق التقييم
        if gaps_probe.get('sensitive') or planning_probe.get('destructive'):
            return 'CRITICAL'
        elif gaps_probe.get('requires_db') and not planning_probe.get('db_ready'):
            return 'HIGH'
        else:
            return 'LOW'
```
